Post-Soviet Gravity Replication in Python

Purpose

This notebook is the Python replication workflow for the Post-Soviet gravity project. It inspects the cleaned dataset, prepares gravity variables, estimates the baseline model sequence, and creates tables for comparison with the manuscript.

The notebook expects the real dataset at:

data/ExSoviet_balanced_clean.csv

It does not create synthetic data and it does not use log(flow + 1).

Prepare gravity variables

The log-linear models use positive trade flows only. The PPML models use flow in levels and can include zero flows if the explanatory variables are valid.

def prepare_gravity_variables(df):
    data = df.copy()

    numeric_columns = [
        "flow",
        "gdp_o",
        "gdp_d",
        "distw",
        "comlang_off",
        "contig",
        "wto_joint",
        "EU_joint",
        "EAEU_joint",
        "year",
    ]

    for column in numeric_columns:
        data[column] = pd.to_numeric(data[column], errors="coerce")

    data["iso_o"] = data["iso_o"].astype(str)
    data["iso_d"] = data["iso_d"].astype(str)
    data["year"] = data["year"].astype("Int64")

    data["pair_id"] = data["iso_o"] + "_" + data["iso_d"]
    data["exporter_year"] = data["iso_o"] + "_" + data["year"].astype(str)
    data["importer_year"] = data["iso_d"] + "_" + data["year"].astype(str)

    data["log_flow"] = np.where(data["flow"] > 0, np.log(data["flow"]), np.nan)
    data["log_gdp_o"] = np.where(data["gdp_o"] > 0, np.log(data["gdp_o"]), np.nan)
    data["log_gdp_d"] = np.where(data["gdp_d"] > 0, np.log(data["gdp_d"]), np.nan)
    data["log_distw"] = np.where(data["distw"] > 0, np.log(data["distw"]), np.nan)

    data["log_trade_intensity"] = np.where(
        (data["flow"] > 0) & (data["gdp_o"] > 0) & (data["gdp_d"] > 0),
        np.log(data["flow"]) - np.log(data["gdp_o"]) - np.log(data["gdp_d"]),
        np.nan,
    )

    return data

Model functions

All OLS models use HC1 robust standard errors. PPML and GPML use robust standard errors where statsmodels supports them.

def model_status(model):
    return {
        "converged": getattr(model, "converged", np.nan),
        "nobs": getattr(model, "nobs", np.nan),
        "llf": getattr(model, "llf", np.nan),
    }


def double_demean(data, variable, origin_col="iso_o", destination_col="iso_d"):
    overall = data[variable].mean()
    origin_mean = data.groupby(origin_col)[variable].transform("mean")
    destination_mean = data.groupby(destination_col)[variable].transform("mean")
    return data[variable] - origin_mean - destination_mean + overall


def coefficient_table(model, model_name, variables=KEY_COEFFICIENTS):
    rows = []
    for variable in variables:
        if variable in model.params.index:
            rows.append(
                {
                    "model": model_name,
                    "variable": variable,
                    "coefficient": model.params[variable],
                    "standard_error": model.bse[variable],
                    "p_value": model.pvalues[variable],
                }
            )
        else:
            rows.append(
                {
                    "model": model_name,
                    "variable": variable,
                    "coefficient": np.nan,
                    "standard_error": np.nan,
                    "p_value": np.nan,
                }
            )
    return pd.DataFrame(rows)

def python_beta_for(model_name, variable):
    model = comparison_models[model_name]
    parameter = ddm_parameter_map.get(variable, variable) if model_name == "DDM" else variable

    if parameter not in model.params.index:
        return np.nan

    return model.params[parameter]


def classify_match(difference):
    if pd.isna(difference):
        return "not_applicable"

    absolute_difference = abs(difference)
    if absolute_difference < 0.001:
        return "exact"
    if absolute_difference < 0.01:
        return "close"
    return "review"

comparison_rows = []
for row in manuscript_coefficients.itertuples(index=False):

        python_beta - manuscript_beta

Discrepancy diagnosis

If Python estimates differ from the manuscript, check these sources first:

  • sample mismatch;
  • zero-flow treatment;
  • fixed effects;
  • robust versus clustered standard errors;
  • variable definitions;
  • missing values;
  • scaling of trade or GDP;
  • differences between manuscript tables and the exact code used to produce them.
